```bash

  --modality-config-path ../../isaaclab_arena_gr00t/embodiments/g1/g1_sim_wbc_data_config.py \
  --model-path $MODELS_DIR/checkpoint-20000 \
  --embodiment-tag NEW_EMBODIMENT \
  --device cuda --host 127.0.0.1 --port 5556
```
